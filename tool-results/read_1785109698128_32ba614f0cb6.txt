   289→function DeploymentPipeline({ deployment }: { deployment: Deployment }) {
   290→  const { t, locale } = useI18n();
   291→  const stageLabels: Record<string, string> = {
   292→    queued: t("deployments.stage.queued"),
   293→    cloning: t("deployments.stage.cloning"),
   294→    building: t("deployments.stage.building"),
   295→    pushing: t("deployments.stage.pushing"),
   296→    starting: t("deployments.stage.starting"),
   297→    health: t("deployments.stage.health"),
   298→    done: t("deployments.stage.done"),
   299→  };
   300→
   301→  return (
   302→    <div className="space-y-5">
   303→      <div className="flex items-start justify-between">
   304→        <div>
   305→          <div className="flex items-center gap-2 mb-1">
   306→            <code className="text-sm text-violet-300 bg-violet-500/10 px-2 py-0.5 rounded">{deployment.commitSha}</code>
   307→            <StatusBadge status={deployment.status} />
   308→          </div>
   309→          <p className="text-sm">{deployment.commitMessage}</p>
   310→          <p className="text-xs text-muted-foreground mt-1" suppressHydrationWarning>
   311→            {deployment.author} · {deployment.branch} · {timeAgo(deployment.startedAt, locale)}
   312→            {deployment.durationMs && ` · ${formatDuration(deployment.durationMs, locale)}`}
   313→          </p>
   314→        </div>
   315→        {deployment.url && (
   316→          <a href={deployment.url} target="_blank" rel="noreferrer">
   317→            <Button variant="outline" size="sm">
   318→              <ExternalLink className="w-3 h-3" />
   319→              Visit
   320→            </Button>
   321→          </a>
   322→        )}
   323→      </div>
   324→
   325→      {deployment.errorMessage && (
   326→        <div className="p-3 rounded-lg bg-rose-500/10 border border-rose-500/20 flex items-start gap-2">
   327→          <AlertTriangle className="w-4 h-4 text-rose-400 shrink-0 mt-0.5" />
   328→          <div>
   329→            <p className="text-xs font-medium text-rose-300">Build failed</p>
   330→            <p className="text-xs text-rose-300/70 mt-0.5 font-mono">{deployment.errorMessage}</p>
   331→          </div>
   332→        </div>
   333→      )}
   334→
   335→      <div>
   336→        <h4 className="text-xs font-semibold uppercase tracking-wider text-muted-foreground mb-3">{t("deployments.pipeline")}</h4>
   337→        <div className="space-y-2">
   338→          {deployment.stages.map((stage, i) => {
   339→            const label = stageLabels[stage.id] || stage.id;
   340→            return (
   341→              <div key={stage.id} className="flex items-center gap-3">
   342→                <div className={cn(
   343→                  "w-6 h-6 rounded-full flex items-center justify-center text-[10px] font-semibold shrink-0",
   344→                  stage.status === "success" && "bg-emerald-500/20 text-emerald-300",
   345→                  stage.status === "running" && "bg-violet-500/20 text-violet-300",
   346→                  stage.status === "failed" && "bg-rose-500/20 text-rose-300",
   347→                  stage.status === "pending" && "bg-white/5 text-muted-foreground",
   348→                  stage.status === "skipped" && "bg-white/5 text-muted-foreground/50"
   349→                )}>
   350→                  {stage.status === "success" && "✓"}
   351→                  {stage.status === "failed" && "✗"}
   352→                  {stage.status === "running" && <Circle className="w-2 h-2 fill-violet-300 animate-pulse" />}
   353→                  {(stage.status === "pending" || stage.status === "skipped") && i + 1}
   354→                </div>
   355→                <div className="flex-1 flex items-center justify-between">
   356→                  <span className={cn(
   357→                    "text-sm",
   358→                    stage.status === "pending" || stage.status === "skipped" ? "text-muted-foreground/60" : "text-foreground"
   359→                  )}>
   360→                    {label}
   361→                  </span>
   362→                  {stage.status === "running" && (
   363→                    <span className="text-[10px] text-violet-300 flex items-center gap-1">
   364→                      <span className="w-1 h-1 rounded-full bg-violet-300 animate-pulse" />
   365→                      running…
   366→                    </span>
   367→                  )}
   368→                </div>
   369→              </div>
   370→            );
   371→          })}
   372→        </div>
   373→      </div>
   374→    </div>
   375→  );
   376→}

   434→function EnvironmentTab() {
   435→  const { t } = useI18n();
   436→  const [vars, setVars] = React.useState(mockEnvVariables);
   437→  const [revealed, setRevealed] = React.useState<Set<string>>(new Set());
   438→  const [copied, setCopied] = React.useState<string | null>(null);
   439→
   440→  const toggleReveal = (id: string) => {
   441→    setRevealed((s) => {
   442→      const n = new Set(s);
   443→      if (n.has(id)) n.delete(id); else n.add(id);
   444→      return n;
   445→    });
   446→  };
   447→
   448→  const copy = (id: string, value: string) => {
   449→    navigator.clipboard?.writeText(value);
   450→    setCopied(id);
   451→    setTimeout(() => setCopied(null), 1500);
   452→  };
   453→
   454→  return (
   455→    <div className="glass-card p-5">
   456→      <SectionHeader
   457→        title={t("project.environment")}
   458→        action={
   459→          <Button size="sm" className="bg-gradient-to-r from-violet-500 to-cyan-500 border-0">

   495→function LogsTab({ projectName }: { projectName: string }) {
   496→  const { t } = useI18n();
   497→  const [logs, setLogs] = React.useState<LogEntry[]>([]);
   498→  React.useEffect(() => { setLogs(generateLogs(50)); }, []);
   499→  const [filter, setFilter] = React.useState("");
   500→  const [autoScroll, setAutoScroll] = React.useState(true);
   501→  const containerRef = React.useRef<HTMLDivElement>(null);
   502→
   503→  useInterval(() => {
   504→    const newLog = generateLogs(1)[0];
   505→    setLogs((l) => [newLog, ...l].slice(0, 500));
   506→  }, 4000);
   507→
   508→  React.useEffect(() => {
   509→    if (autoScroll && containerRef.current) {
   510→      containerRef.current.scrollTop = containerRef.current.scrollHeight;
   511→    }
   512→  }, [logs, autoScroll]);
   513→
   514→  const filtered = logs.filter((l) =>
   515→    !filter ||
   516→    l.message.toLowerCase().includes(filter.toLowerCase()) ||
   517→    l.source.toLowerCase().includes(filter.toLowerCase())
   518→  );
   519→
   520→  const levelColor: Record<LogEntry["level"], string> = {
   521→    info: "text-sky-300",
   522→    warn: "text-amber-300",
   523→    error: "text-rose-300",
   524→    debug: "text-muted-foreground",
   525→    success: "text-emerald-300",
   526→  };
   527→
   528→  return (
   529→    <div className="space-y-3">
   530→      <div className="flex items-center gap-2">
   531→        <div className="relative flex-1">
   532→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   533→          <Input
   534→            value={filter}
   535→            onChange={(e) => setFilter(e.target.value)}
   536→            placeholder={t("project.logs.search")}
   537→            className="pl-9 bg-white/5 border-white/10 font-mono text-sm"
   538→          />
   539→        </div>
   540→        <Button variant="outline" size="sm" onClick={() => setLogs([])}>
   541→          {t("project.logs.clear")}
   542→        </Button>
   543→        <Button variant="outline" size="sm">
   544→          <Download className="w-3.5 h-3.5" />
   545→          {t("project.logs.download")}
   546→        </Button>
   547→      </div>
   548→
   549→      <div className="flex items-center justify-between px-1">
   550→        <div className="flex items-center gap-2 text-xs text-muted-foreground">
   551→          <span className="relative inline-flex">
   552→            <span className="inline-block w-1.5 h-1.5 rounded-full bg-emerald-400" />
   553→            <span className="absolute inset-0 rounded-full animate-ping opacity-60 bg-emerald-400" />
   554→          </span>
   555→          {t("project.logs.live")} · {projectName}
   556→        </div>
   557→        <label className="flex items-center gap-1.5 text-xs text-muted-foreground cursor-pointer">
   558→          <Switch checked={autoScroll} onCheckedChange={setAutoScroll} />
   559→          Auto-scroll
   560→        </label>
   561→      </div>
   562→
   563→      <div ref={containerRef} className="terminal rounded-lg p-4 max-h-[600px] overflow-y-auto scrollbar-thin text-xs leading-relaxed">
   564→        {filtered.length === 0 ? (
   565→          <div className="text-muted-foreground text-center py-8">{t("common.noData")}</div>
   566→        ) : (
   567→          filtered.map((log) => (
   568→            <div key={log.id} className="flex items-start gap-3 py-0.5 hover:bg-white/[0.02] -mx-2 px-2 rounded">
   569→              <span className="text-muted-foreground/60 shrink-0 tabular-nums">
   570→                {new Date(log.timestamp).toLocaleTimeString("en-GB", { hour12: false })}
   571→              </span>
   572→              <span className={cn("shrink-0 font-semibold uppercase w-12", levelColor[log.level])}>
   573→                {log.level}
   574→              </span>
   575→              <span className="text-muted-foreground/70 shrink-0">[{log.source}]</span>
   576→              <span className="flex-1 break-all">{log.message}</span>
   577→            </div>
   578→          ))
   579→        )}
   580→      </div>
   581→    </div>
   582→  );
